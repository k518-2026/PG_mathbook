\chapter*{はじめに：科学は仮説の上に立っている}
\addcontentsline{toc}{chapter}{はじめに：科学は仮説の上に立っている}
\markboth{はじめに}{はじめに}
%==========================================================

『科学と仮説』（1902年）は、数学者アンリ・ポアンカレ（1854〜1912）が、一般の読者に向けて書いた本です。「数学や物理学の基礎には、どんな仮定（仮説）があるのか」という問いを、4つの部（数・空間・力・自然）で考えます。本書も、翻訳版の13章の順に進みます。

\begin{figure}[htbp]
\centering
\begin{tikzpicture}[node distance=4mm]
 \node[fbox, text width=3.1cm] (a) {第1部 数と量\\{\small 第1〜2章}};
 \node[fbox, text width=3.1cm, right=of a] (b) {第2部 空間\\{\small 第3〜5章}};
 \node[fbox, text width=3.1cm, right=of b] (c) {第3部 力\\{\small 第6〜8章}};
 \node[fbox, text width=3.1cm, right=of c] (d) {第4部 自然\\{\small 第9〜13章}};
 \draw[ax] (a)--(b); \draw[ax] (b)--(c); \draw[ax] (c)--(d);
\end{tikzpicture}
\caption{本書の4部構成。基礎から、物理学の理論へと進む。}
\end{figure}

\begin{notebox}{事実・推測・未確認の区別}
ポアンカレの哲学（規約主義）は、要約して紹介します。数値例は、筆者が計算して確かめたものです。翻訳版は小型モデルによる訳であり、細部は原著との照合が必要です。
\end{notebox}

%==========================================================
\chapter{数学的推論の性質}
\markboth{第1章}{数学的推論の性質}
%==========================================================

ポアンカレは、数学の推論には、論理だけでは尽くせない、本質的な方法があると考えました。それが、\textbf{数学的帰納法}（漸化的推論）です。

\begin{figure}[htbp]
\centering
\begin{tikzpicture}
 \foreach \i in {1,...,8} {
  \draw[thick, primaryblue, fill=softblue] (\i*0.9,0) rectangle (\i*0.9+0.35,1.1);
  \node[font=\scriptsize] at (\i*0.9+0.17,-0.25) {\i};
 }
 \draw[thick, crimson, ->] (1.4,1.3) to[bend left=25] (2.1,1.3);
 \node[font=\small, text width=7cm] at (4.5,2.1) {1番目が倒れる（$n=1$ で成り立つ）。\\$n$ 番目が倒れれば $n+1$ 番目も倒れる。\\$\Rightarrow$ すべてが倒れる。};
\end{tikzpicture}
\caption{数学的帰納法は、ドミノ倒し。最初の1枚と、「次へ伝わる」ことを確かめれば、無限に続く。}
\end{figure}

\begin{examplebox}{例：1からnまでの和}
$n=1$：左辺 $1$、右辺 $\frac{1\cdot2}{2}=1$ ✓。\\
$n$ で成り立つと仮定。$n+1$ のとき、$1+\cdots+n+(n+1)=\frac{n(n+1)}{2}+(n+1)=\frac{(n+1)(n+2)}{2}$ ✓。
\end{examplebox}

\begin{figure}[htbp]
\centering
\begin{tikzpicture}[scale=0.55]
 \foreach \i in {1,...,5} {
  \foreach \j in {1,...,\i} {
   \fill[softblue, draw=primaryblue] (\j,-\i) rectangle (\j+1,-\i+1);
  }
 }
 \foreach \i in {1,...,5} {\node[font=\scriptsize] at (-0.3,-\i+0.5) {\i};}
 \node[font=\footnotesize, text width=6cm] at (11,-2.5) {段の石の数 $1,2,3,4,5$。\\合計 $15=\frac{5\cdot6}{2}$。};
\end{tikzpicture}
\caption{階段状の石の数え方。$1+2+3+4+5=15$。}
\end{figure}

\begin{pointbox}{この章のポイント}
\begin{itemize}[leftmargin=1.5em]
\item 帰納法は、「最初」と「$n\to n+1$」の2つで無限を扱う。
\item ポアンカレは、これを数学に固有の、創造的な推論と見た。
\end{itemize}
\end{pointbox}

%==========================================================
\chapter{数学的量と経験}
\markboth{第2章}{数学的量と経験}
%==========================================================

私たちの感覚は、細かい差を区別できません。ポアンカレは、これを「物理的連続体」と呼び、数学の連続体と区別しました。

\begin{figure}[htbp]
\centering
\begin{tikzpicture}
 \draw[ax] (0,0)--(8,0);
 \foreach \xa/\lab in {1/10 g,3/11 g,5/12 g} {\fill[primaryblue] (\xa,0) circle (0.1); \node[below] at (\xa,-0.2) {\lab};}
 \draw[<->, accentgreen] (1,0.5)--(3,0.5) node[midway, above, font=\footnotesize] {区別できない $A=B$};
 \draw[<->, accentgreen] (3,1.2)--(5,1.2) node[midway, above, font=\footnotesize] {区別できない $B=C$};
 \draw[<->, crimson] (1,2.0)--(5,2.0) node[midway, above, font=\footnotesize] {区別できる $A<C$};
\end{tikzpicture}
\caption{感覚の連続体。区別できる最小の差を1.5 gとすると、10 gと11 g、11 gと12 gは同じに感じるが、10 gと12 gは区別できる。}
\end{figure}

つまり、感覚の世界では、$A=B$、$B=C$ でも、$A=C$ とは限りません。数学では、こうした矛盾を避けるため、新しい「数の連続体」を作りました。

\begin{figure}[htbp]
\centering
\begin{tikzpicture}[node distance=8mm]
 \node[wbox, text width=3cm] (a) {感覚の連続体\\{\small $A=B,B=C$ でも\\$A\neq C$}};
 \node[gbox, text width=3.6cm, right=22mm of a] (b) {数の連続体\\{\small $=$ は推移的\\（$A=B,B=C\Rightarrow A=C$）}};
 \draw[ax] (a)--node[above, lbl]{理想化} (b);
\end{tikzpicture}
\caption{経験から数学へ。数学の連続体は、感覚の限界を超えて作られた理想化。}
\end{figure}

\begin{pointbox}{この章のポイント}
\begin{itemize}[leftmargin=1.5em]
\item 感覚の等しさは、推移的ではない。
\item 数学の連続体は、その矛盾を解消した理想化。
\end{itemize}
\end{pointbox}

\chapter{非ユークリッド幾何学}
\markboth{第3章}{非ユークリッド幾何学}
%==========================================================

「平行線の公理」を変えると、別の幾何学ができます。その1つ、双曲幾何学の姿を、ポアンカレの円板模型で見てみましょう。円板の中だけを「世界」と考え、円板の「直線」を、境界の円に直交する円弧とします。

\begin{figure}[htbp]
\centering
\begin{tikzpicture}[scale=3.0]
 \draw[thick, primaryblue, fill=softblue] (0,0) circle (1);
 \begin{scope}
  \clip (0,0) circle (1);
  \foreach \ang in {0,60,120} {
   \begin{scope}[rotate=\ang]
    \draw[thick, crimson] (-1,0)--(1,0);
   \end{scope}
  }
  \foreach \ang in {20,80,140,200,260,320} {
   \begin{scope}[rotate=\ang]
    \draw[thick, accentgreen] (1.6,0) circle ({sqrt(1.6*1.6-1)});
   \end{scope}
  }
 \end{scope}
\end{tikzpicture}
\caption{ポアンカレの円板。円板の「直線」は、直径（赤）と、境界の円に直交する円弧（緑）。円周は無限の遠方にあたる。}
\end{figure}

この世界では、1つの直線 $\ell$ の外の点を通る、$\ell$ と交わらない直線が、無数に引けます。そして、三角形の内角の和は $180^\circ$ より小さくなります。

\begin{figure}[htbp]
\centering
\begin{tikzpicture}[scale=0.9]
 \draw[thick, primaryblue] (0,0)--(5,0) node[right] {$\ell$};
 \fill (2.5,2) circle (0.07) node[above right] {$P$};
 \foreach \s in {-0.6,-0.2,0.2,0.6} {\draw[crimson, thick] (2.5-3,2-3*\s)--(2.5+3,2+3*\s);}
 \node[font=\footnotesize, align=center] at (7.2,1.5) {双曲幾何では、$P$ を通り\\ $\ell$ と交わらない直線が\\ 無数にある};
\end{tikzpicture}
\caption{平行線の公理を変えた世界。$P$ を通って $\ell$ と交わらない直線（赤）が何本も引ける。}
\end{figure}

\begin{notebox}{注意}
図の赤い直線は、ユークリッド平面上に描いたもので、双曲幾何学の世界での姿とは異なります。「交わらない直線が複数ある」という意味の模式図です。
\end{notebox}

\begin{pointbox}{この章のポイント}
\begin{itemize}[leftmargin=1.5em]
\item 平行線の公理を変えると、別の矛盾のない幾何学ができる。
\item ポアンカレの円板は、双曲幾何学を普通の円板の中に写し取る。
\end{itemize}
\end{pointbox}

\section*{もし、世界のすべてが2倍に大きくなったら}

ポアンカレは、次の思考実験をします。ある夜、宇宙のすべての大きさが、物差しも含めて、2倍になったとしたら、私たちは気づくでしょうか。物差しで測った長さは、変わりません。だから、気づくことはできません。

\begin{figure}[htbp]
\centering
\begin{tikzpicture}
 \draw[thick, primaryblue, fill=softblue] (0,0) rectangle (1.2,0.8);
 \draw[thick] (0,-0.4)--(1.2,-0.4) node[midway, below, font=\footnotesize] {物差し 2目盛};
 \draw[ax] (2.4,0.2)--(3.6,0.2);
 \draw[thick, primaryblue, fill=softblue] (4.8,0) rectangle (7.2,1.6);
 \draw[thick] (4.8,-0.4)--(7.2,-0.4) node[midway, below, font=\footnotesize] {物差し（2倍）で 2目盛};
\end{tikzpicture}
\caption{すべてが2倍になっても、物差しで測れば同じ。大きさは相対的で、絶対的ではない。}
\end{figure}

\begin{pointbox}{この章のポイント}
\begin{itemize}[leftmargin=1.5em]
\item 空間は、感覚と運動の経験から作られる。
\item 大きさは相対的。全体が同じ割合で変われば、気づけない。
\end{itemize}
\end{pointbox}

\chapter{経験と幾何学}
\markboth{第5章}{経験と幾何学}
%==========================================================

「三角形の内角の和は $180^\circ$ か」は、実験で確かめられるでしょうか。ポアンカレの答えは、「確かめられるのは、幾何学と物理学（光の進み方）の組み合わせだけ」です。

\begin{figure}[htbp]
\centering
\begin{tikzpicture}[node distance=8mm]
 \node[wbox, text width=4cm] (a) {測定：星を見て\\三角形の角の和を測る};

\end{tikzpicture}
\caption{同じ測定結果に、2通りの説明が可能。どちらを選ぶかは、便利さの問題（規約）とポアンカレは考えた。}

ポアンカレは、ユークリッド幾何学が、いちばん簡単で便利なので、これからも使われ続けるだろう、と述べます。

\begin{pointbox}{この章のポイント}
\begin{itemize}[leftmargin=1.5em]
\item 実験が確かめるのは、幾何学と物理学のセット。
\item どの幾何学を使うかは、簡単さで選ぶ約束（規約）。
\end{itemize}
\end{pointbox}

\begin{notebox}{のちの歴史}
ポアンカレのこの主張は、のちに、アインシュタインの一般相対性理論が、重力で光が曲がり、空間が曲がっていると説明した点で、別の視点を与えられたと言われます。その評価は専門書で確かめてください。
\end{notebox}

\chapter{相対運動と絶対運動}
\markboth{第7章}{相対運動と絶対運動}
%==========================================================

走る電車の中でボールを投げても、止まった電車の中と同じように見えます。どれが「本当に止まっている」のか、絶対的な基準はありません。

\begin{figure}[htbp]
\centering
\begin{tikzpicture}
 \draw[thick, primaryblue, fill=softblue] (0,0) rectangle (3,1.2);
 \draw[->, thick, crimson] (3.2,0.6)--(4.6,0.6) node[right, font=\small] {速さ $v$};
 \fill[accentgreen] (1.5,0.6) circle (0.12);
 \draw[thick] (-0.5,-0.2)--(5,-0.2);
 \node[font=\footnotesize] at (1.5,-0.6) {地面（ホームの人から見て）};
\end{tikzpicture}
\caption{電車の中の人には、ボールは真上に飛んで戻る。ホームの人には、電車の速さ $v$ ぶん、斜めに進んで見える。}
\end{figure}

位置の変換は $x'=x-vt$ と書けます。力学の法則は、この変換で形を変えません（ガリレイの相対性）。

\begin{examplebox}{数値例}
電車が $v=20\ \mathrm{m/s}$ で走っています。電車の中でボールを真上に投げ、2秒後に手元に戻ったとき、ホームの人から見たボールの水平方向の移動距離は $20\times2=40\ \mathrm{m}$。
\end{examplebox}

\begin{pointbox}{この章のポイント}
\begin{itemize}[leftmargin=1.5em]
\item 絶対的な静止はない。運動は相対的。
\item 力学の法則は、等速で動く観測者どうしで同じ形になる。
\end{itemize}
\end{pointbox}

\chapter{エネルギーと熱力学}
\markboth{第8章}{エネルギーと熱力学}

エネルギーは、姿を変えても、総量が変わらない（エネルギー保存則）。ポアンカレは、これが、実験から導かれる法則であるとともに、「エネルギー」という量の定義のしかたでもあることを指摘します。

\begin{figure}[htbp]
\centering
\begin{tikzpicture}[node distance=10mm]
 \node[abox, text width=2.4cm] (a) {位置エネルギー};
 \node[fbox, text width=2.4cm, right=of a] (b) {運動エネルギー};
 \node[rbox, text width=2.4cm, right=of b] (c) {熱};
 \draw[ax, <->] (a)--(b); \draw[ax] (b)--(c);
\end{tikzpicture}
\caption{エネルギーの変換。落とした物体は、位置エネルギーが運動エネルギーに変わり、最後は熱になる。総量は一定。}
\end{figure}

熱機関の効率には上限があります。高温 $T_h$、低温 $T_c$（絶対温度）の間で働く熱機関の最大の効率は $1-\dfrac{T_c}{T_h}$。

\begin{examplebox}{数値例}
$T_h=500$ K、$T_c=300$ K のとき、$1-\frac{300}{500}=0.4$。最大でも40\%。
\end{examplebox}

\begin{figure}[htbp]
\centering
\begin{tikzpicture}
\begin{axis}[width=9cm, height=5.5cm, xlabel={高温側の温度 $T_h$ [K]}, ylabel={最大効率}, xmin=300, xmax=1000, ymin=0, ymax=0.8, grid=major]
 \addplot[thick, primaryblue, domain=300:1000, samples=40] {1-300/x};
 \addplot[only marks, mark=*, color=crimson] coordinates {(500,0.4)};
\end{axis}
\end{tikzpicture}
\caption{低温側を300 Kに固定したときの最大効率。高温側が高いほど効率が上がる。}
\end{figure}

\begin{pointbox}{この章のポイント}
\begin{itemize}[leftmargin=1.5em]
\item エネルギーは姿を変えても総量が保存される。
\item 熱機関の効率には、温度で決まる上限がある。
\end{itemize}
\end{pointbox}

科学者は、実験をして、法則を作ります。そのとき、いくつかの仮説を置きます。ポアンカレは、仮説を役割で分けます。

\begin{center}
\begin{tabular}{lp{8cm}}
\toprule
種類 & 役割\\
\midrule
検証できる仮説 & 実験で確かめられ、否定されることもある\\

\bottomrule

\end{pointbox}

\chapter{現代物理学の理論}
\markboth{第10章}{現代物理学の理論}
%==========================================================

ポアンカレは、同じ現象を説明する理論が複数ありうることに、注目します。たとえば、光や電気について、いくつかの異なる力学的な「模型」が考えられても、方程式が同じなら、予測も同じです。

\begin{figure}[htbp]
\centering
\begin{tikzpicture}[node distance=8mm]
 \node[wbox, text width=2.6cm] (a) {模型A\\{\small 例：流体のモデル}};
 \node[wbox, text width=2.6cm, below=of a] (b) {模型B\\{\small 例：弾性体のモデル}};
 \node[gbox, text width=3.2cm, right=26mm of $(a)!0.5!(b)$] (c) {同じ方程式};
 \node[fbox, text width=3.2cm, right=18mm of c] (d) {同じ予測};
 \draw[ax] (a)--(c); \draw[ax] (b)--(c); \draw[ax] (c)--(d);
\end{tikzpicture}
\caption{異なる模型でも、方程式が同じなら、予測は同じ。どちらが「本物」かは、実験では決められない。}
\end{figure}

\begin{notebox}{注意}
具体的なモデルの例（流体・弾性体）は、考え方を示す例示であり、ポアンカレが挙げた物理的な模型そのものではありません。
\end{notebox}

\begin{pointbox}{この章のポイント}
\begin{itemize}[leftmargin=1.5em]
\item 理論は、方程式の関係を教える。模型は、理解の助けにすぎない。
\end{itemize}
\end{pointbox}

\chapter{確率論の計算}
\markboth{第11章}{確率論}
%==========================================================

ポアンカレは、ルーレットを例に、「偶然」を説明します。わずかな初期条件の違い（回す速さ）が、結果（赤か黒か）を大きく変えます。

\begin{figure}[htbp]
\centering
\begin{tikzpicture}[scale=0.9]
 \foreach \i in {0,...,35} {
  \pgfmathsetmacro{\a}{10*\i}
  \pgfmathsetmacro{\b}{10*(\i+1)}
  \ifodd\i
    \fill[crimson] (0,0) -- (\a:2) arc (\a:\b:2) -- cycle;
  \else
    \fill[black!75] (0,0) -- (\a:2) arc (\a:\b:2) -- cycle;
  \fi
 }
 \draw[thick] (0,0) circle (2);
\end{tikzpicture}
\caption{細かく赤と黒に分けたルーレット。回す速さがほんの少し違うだけで、止まる色が変わる。}
\end{figure}

\begin{figure}[htbp]
\centering
\begin{tikzpicture}
 \foreach \i in {0,...,39} {
  \pgfmathsetmacro{\xa}{\i*0.3}
  \ifodd\i
    \fill[crimson] (\xa,0) rectangle (\xa+0.3,0.8);
  \else
    \fill[black!75] (\xa,0) rectangle (\xa+0.3,0.8);
  \fi
 }
 \draw[ax] (0,-0.2)--(12.3,-0.2) node[right, font=\footnotesize] {初速};
 \node[font=\footnotesize] at (6,1.3) {初速が少し違うと、結果の色が入れ替わる};
\end{tikzpicture}
\caption{初速と結果の色の対応（模式図）。帯が細かいほど、結果は事実上予測できず、赤と黒は半々になる。}
\end{figure}

初速の幅がある程度ばらつくとき、赤と黒の帯がきわめて細ければ、赤になる確率は $\frac12$ に近くなります。

\begin{pointbox}{この章のポイント}
\begin{itemize}[leftmargin=1.5em]
\item 偶然とは、小さな原因が大きな結果を生むこと。
\item 帯が細かいと、確率は帯の幅の比（半々）で決まる。
\end{itemize}
\end{pointbox}

\chapter{光学と電磁気学}
\markboth{第12章}{光学と電磁気学}
%==========================================================

19世紀に、光が電磁気の波であることが分かりました（マクスウェルの理論）。電場と磁場が、互いに垂直に振動しながら進みます。

\begin{figure}[htbp]
\centering
\begin{tikzpicture}[scale=1.0]
 \draw[->] (0,0)--(10,0) node[right] {進行方向};
 \draw[thick, primaryblue, domain=0:9.4, samples=100] plot (\x, {1.0*sin(\x*57.2958*1.0)});
 \draw[thick, crimson, dashed, domain=0:9.4, samples=100] plot (\x, {0.55*sin(\x*57.2958*1.0)});
 \node[primaryblue, font=\footnotesize] at (1.6,1.3) {電場 $E$};
 \node[crimson, font=\footnotesize] at (4.7,-1.3) {磁場 $B$（電場と垂直な面）};
\end{tikzpicture}
\caption{電磁波の模式図。電場（青）と磁場（赤、実際は電場と垂直な向きで振動）が、進行方向に垂直に振動する。}
\end{figure}

\begin{notebox}{図についての注意}
2次元の図では、磁場が電場と垂直な向きであることを正確に描けません。赤い破線は、磁場が電場と同時に同じ位相で振動することを示す記号的な表現で、実際の向きは紙面に垂直と考えてください。
\end{notebox}

\begin{pointbox}{この章のポイント}
\begin{itemize}[leftmargin=1.5em]
\item 光は電磁波。電場と磁場が垂直に振動して進む。
\end{itemize}
\end{pointbox}

\chapter{電気力学とローレンツ}
\markboth{第13章}{電気力学とローレンツ}
%==========================================================

電気を帯びた粒子の運動を調べるうちに、動いている物体では、時間や長さが、ふつうの感覚と少し違うことが分かってきました。ローレンツは、速さ $v$ で動く系のために、係数 $\gamma=\dfrac{1}{\sqrt{1-v^2/c^2}}$ を導入しました。

\begin{center}
\begin{tabular}{rr}
\toprule
$v/c$ & $\gamma$\\
\midrule
0.5 & 1.155\\
0.9 & 2.294\\
0.99 & 7.089\\
\bottomrule
\end{tabular}
\end{center}

\begin{figure}[htbp]
\centering
\begin{tikzpicture}

\begin{notebox}{この本の未解決点}
\begin{itemize}[leftmargin=1.5em]
\item ポアンカレの規約主義の評価は、本書では行いません。
\item ローレンツ因子 $\gamma$ の値は、公式から筆者が計算しました。
\item 翻訳版の細部は、原著との照合が必要です。
\end{itemize}
\end{notebox}
