\section{3つの記号を怖がらない}

微積分でいちばん最初に立ちはだかるのは、$\dd$ と $\int$ という記号です。トンプソンは、まずここを「怖くない」と言います。

\begin{center}
\begin{tabular}{lp{9cm}}
\toprule
記号 & 意味 \\
\midrule
$\dd x$ & $x$ の「ほんの少しの量」。（$\dd$ は「ほんのわずかな」の意味で、掛け算ではない）\\
$\displaystyle\frac{\dd y}{\dd x}$ & $x$ をほんの少し増やしたとき、$y$ がどれだけ増えるかの比\\
$\displaystyle\int$ & 「たくさんの小さな量を足し合わせる」という意味の記号（Sum の S を縦に伸ばした形）\\
\bottomrule
\end{tabular}
\end{center}

\begin{figure}[htbp]
\centering
\begin{tikzpicture}[scale=1.0]
 \draw[->] (0,0)--(6,0) node[right] {$x$}; \draw[->] (0,0)--(0,4) node[above] {$y$};
 \draw[thick, primaryblue, domain=0.3:5.2, samples=40] plot (\x, {0.12*\x*\x+0.4});
 \coordinate (P) at (2.5,{0.12*2.5*2.5+0.4});
 \coordinate (Q) at (3.5,{0.12*3.5*3.5+0.4});
 \draw[dashed] (P) -- (3.5, {0.12*2.5*2.5+0.4}) -- (Q);
 \fill[crimson] (P) circle (0.06); \fill[crimson] (Q) circle (0.06);
 \node[below, font=\small] at (3.0,{0.12*2.5*2.5+0.4}) {$\dd x$};
 \node[right, font=\small] at (3.5,{0.12*3.1*3.1+0.45}) {$\dd y$};
\end{tikzpicture}
\caption{$\dd x$ は横に少し進んだ量、$\dd y$ はそのとき縦に増えた量。$\dd y/\dd x$ は、その部分の曲線の傾き。（図では見やすいよう大きく描いている）}
\end{figure}

\section{積分記号は「足し合わせ」}

長さを細かい断片 $\dd x$ に分けて、全部を足し合わせたものが全体の長さ。そのことを $\int \dd x$ と書きます。たとえば、1分の$\frac{1}{60}$（つまり1秒）を60個足すと1分です。

\begin{figure}[htbp]
\centering
\begin{tikzpicture}
 \foreach \xa in {0,0.5,...,4.5} {\draw[thick, fill=softblue, draw=primaryblue] (\xa,0) rectangle (\xa+0.5,0.5);}
 \draw[decorate, decoration={brace, mirror, amplitude=5pt}] (0,-0.1)--(5,-0.1) node[midway, below=6pt, font=\small] {$\displaystyle\int \dd x$ ＝ 小さな $\dd x$ をすべて足したもの};
\end{tikzpicture}
\caption{積分は、細切れの合計。（10個の断片の例）}
\end{figure}

\begin{pointbox}{この章のポイント}
\begin{itemize}[leftmargin=1.5em]
\item $\dd x$ は「ほんの少しの $x$」。$\dd y/\dd x$ は傾き。$\int$ は足し合わせ。
\item 記号を読めるようになるだけで、半分は怖くなくなる。
\end{itemize}
\end{pointbox}

\section{小さいものの、さらに小さいもの}

1分の$\frac1{60}$は1秒。1秒の$\frac1{60}$をさらにとると、1分の$\frac1{3600}$。それくらい小さなものは、1分という大きさに比べれば、ほとんど無視できます。

\begin{figure}[htbp]
\centering
\begin{tikzpicture}[node distance=10mm]
 \node[fbox, text width=2.6cm] (a) {1分};
 \node[fbox, text width=2.6cm, right=of a] (b) {1秒\\{\small 1分の $\frac1{60}$}};
 \node[fbox, text width=2.6cm, right=of b] (c) {1秒の$\frac1{60}$\\{\small 1分の $\frac1{3600}$}};
 \draw[ax] (a)--node[above, lbl]{$\times\frac1{60}$} (b);
 \draw[ax] (b)--node[above, lbl]{$\times\frac1{60}$} (c);
\end{tikzpicture}
\caption{小ささの階級。「小さいものの小さいもの」は、もう一段小さい。}
\end{figure}

\section{$x$ を少し増やすと $x^2$ はどう変わるか}

1辺 $x$ の正方形の面積は $x^2$。1辺をほんの少し $\dd x$ だけ増やすと、面積は $(x+\dd x)^2=x^2+2x\,\dd x+(\dd x)^2$。

\begin{figure}[htbp]
\centering
\begin{tikzpicture}[scale=0.9]
 \draw[thick, fill=softblue] (0,0) rectangle (4,4);
 \draw[thick, fill=softamber] (4,0) rectangle (4.7,4);
 \draw[thick, fill=softamber] (0,4) rectangle (4,4.7);
 \draw[thick, fill=softrose] (4,4) rectangle (4.7,4.7);
 \node at (2,2) {$x^2$};
 \node[font=\scriptsize] at (4.35,2) {$x\,\dd x$};
 \node[font=\scriptsize] at (2,4.35) {$x\,\dd x$};
 \node[font=\tiny, rotate=0] at (5.5,4.35) {$(\dd x)^2$};
 \draw[->] (5.1,4.35)--(4.72,4.35);
\end{tikzpicture}
\caption{正方形の辺を $\dd x$ だけ増やしたときの面積の増え方。2本の細長い帯が $2x\,\dd x$、すみの小さな正方形が $(\dd x)^2$。（見やすいよう $\dd x$ を大きく描いている）}
\end{figure}

$(\dd x)^2$ は、$2x\,\dd x$ に比べてさらに小さいので、無視します。残るのは $2x\,\dd x$。これが、「増えた面積 $\dd y$」で、$\dd y/\dd x=2x$ となります。

\begin{examplebox}{数値で確かめる}
$x=10$ から $x=10.01$ に増やす。$\dd x=0.01$。\\
$y=x^2$ は、$100$ から $100.2001$ へ。増加は $0.2001$。\\
$2x\,\dd x=2\times10\times0.01=0.2$。差は $0.0001=(\dd x)^2$ で、増加分の$\frac1{2000}$。
\end{examplebox}

\begin{pointbox}{この章のポイント}
\begin{itemize}[leftmargin=1.5em]
\item 小さいものの2乗は、さらに小さく、無視できる。
\item $y=x^2$ なら $\dd y=2x\,\dd x$。
\end{itemize}
\end{pointbox}

\section{「どれだけ速く増えるか」の比}

$y=2x$ なら、$x$ が1増えると $y$ は2増えます。$\dd y/\dd x=2$ は、「$y$ は $x$ の2倍の速さで増える」という意味です。

\centering
\begin{tikzpicture}
\begin{axis}[width=9cm, height=5.5cm, xlabel={$x$}, ylabel={$y$}, xmin=0, xmax=4, ymin=0, ymax=8, grid=major, legend pos=north west, legend style={font=\footnotesize}]
 \addplot[thick, primaryblue, domain=0:4] {2*x};
 \addplot[thick, crimson, domain=0:4] {x};
 \legend{$y=2x$（傾き2）,$y=x$（傾き1）}
\end{axis}
\end{tikzpicture}
\caption{傾きは、$x$ が1増えたときに $y$ がいくつ増えるか。$y=2x$ は $y=x$ の2倍の急さ。}
\end{figure}

\section{$y=x^2$ の増え方は場所によって違う}

$y=x^2$ では、$\dd y/\dd x=2x$。$x=1$ では傾き2、$x=3$ では傾き6、$x=5$ では傾き10。右へ行くほど、急に増えていきます。

\begin{figure}[htbp]
\centering
\begin{tikzpicture}
\begin{axis}[width=9.5cm, height=6cm, xlabel={$x$}, ylabel={$y=x^2$}, xmin=0, xmax=5, ymin=0, ymax=25, grid=major]

 \addplot[only marks, mark=*, color=crimson] coordinates {(1,1)(3,9)(5,25)};
\end{axis}
\end{tikzpicture}
\caption{$y=x^2$ と、$x=1,3,5$ での接線。傾きはそれぞれ $2, 6, 10$。}
\end{figure}

\begin{pointbox}{この章のポイント}
\begin{itemize}[leftmargin=1.5em]
\item $\dd y/\dd x$ は、「$x$ に対する $y$ の増え方の速さ」。
\item $y=x^2$ の速さは $2x$。場所で変わる。
\end{itemize}
\end{pointbox}

\section{べき乗の微分}

$y=x^n$ のとき、

\[
\frac{\dd y}{\dd x}=n\,x^{n-1}.
\]

$x^3$ で確かめます。$(x+\dd x)^3=x^3+3x^2\dd x+3x(\dd x)^2+(\dd x)^3$。小さいものの2乗・3乗を捨てると、増加は $3x^2\dd x$。ゆえに $\dd y/\dd x=3x^2$。

\begin{figure}[htbp]
\centering
\begin{tikzpicture}[x={(1cm,0cm)}, y={(0cm,1cm)}, z={(0.5cm,0.4cm)}, scale=0.9]
 \draw[thick, fill=softblue] (0,0,0) -- (3,0,0) -- (3,3,0) -- (0,3,0) -- cycle;
 \draw[thick, fill=softblue!70!white] (0,3,0) -- (3,3,0) -- (3,3,3) -- (0,3,3) -- cycle;
 \draw[thick, fill=softblue!50!white] (3,0,0) -- (3,0,3) -- (3,3,3) -- (3,3,0) -- cycle;
 \draw[thick, fill=softamber] (3,0,0) -- (3.5,0,0) -- (3.5,3,0) -- (3,3,0) -- cycle;
 \draw[thick, fill=softamber] (0,3,0) -- (3,3,0) -- (3,3.5,0) -- (0,3.5,0) -- cycle;
 \node at (1.5,1.5,0) {$x^3$};
 \node[font=\footnotesize, text width=5cm] at (8,1.5) {各面に厚さ $\dd x$ の板を貼る：\\3枚の板の体積が $3x^2\,\dd x$};
\end{tikzpicture}
\caption{立方体の辺を $\dd x$ 増やすと、3つの面に板が増える。体積の増加は主に $3x^2\dd x$。（辺・すみの小さな部分は省略）}
\end{figure}

\begin{center}
\begin{tabular}{ll}
\toprule
$y$ & $\dd y/\dd x$\\
\midrule
$x$ & $1$\\
$x^2$ & $2x$\\
$x^3$ & $3x^2$\\
$x^4$ & $4x^3$\\
$\sqrt{x}=x^{1/2}$ & $\frac12x^{-1/2}=\dfrac{1}{2\sqrt x}$\\
$\dfrac1x=x^{-1}$ & $-x^{-2}=-\dfrac1{x^2}$\\
\bottomrule
\end{tabular}
\end{center}

\centering
\begin{tikzpicture}

\caption{$\sqrt x$ と $1/x$ と、それぞれの微分のグラフ。$1/x$ は減少関数なので、その傾き $-1/x^2$ は負。}
\end{figure}

\begin{pointbox}{この章のポイント}
\begin{itemize}[leftmargin=1.5em]
\item $x^n\to nx^{n-1}$。指数が負でも分数でも成り立つ。
\end{itemize}
\end{pointbox}

\chapter{次の段階：定数の扱い方}
\markboth{第5章}{定数の扱い方}

\section{足す定数は消え、かける定数は残る}

$y=x^2+5$ の微分は $2x$。「5」を足しても、変化の仕方は変わらないからです。一方、$y=3x^2$ の微分は $6x$。「3」は微分のあとも残ります。

\begin{figure}[htbp]
\centering
\begin{tikzpicture}
\begin{axis}[width=9.5cm, height=6cm, xlabel={$x$}, ylabel={$y$}, xmin=-3, xmax=3, ymin=0, ymax=16, grid=major, legend pos=north west, legend style={font=\footnotesize}]
 \addplot[thick, primaryblue, domain=-3:3, samples=40] {x^2};
 \addplot[thick, crimson, domain=-3:3, samples=40] {x^2+5};
 \addplot[thick, accentgreen, dashed, domain=-2.3:2.3, samples=40] {3*x^2};
 \legend{$x^2$,$x^2+5$（上へ平行移動）,$3x^2$（縦に3倍）}
\end{axis}
\end{tikzpicture}
\caption{定数を足すと、グラフは上下に平行移動するだけ（傾きは同じ）。定数をかけると縦に伸び、傾きも3倍になる。}
\end{figure}

\begin{examplebox}{例}
$y=ax^3+b$（$a, b$ は定数）：$\dd y/\dd x=3ax^2$。$b$ は消え、$a$ は残る。
\end{examplebox}

\begin{pointbox}{この章のポイント}
\begin{itemize}[leftmargin=1.5em]
\item 足す定数は消える（傾きは変わらない）。
\item かける定数はそのまま残る。
\end{itemize}
\end{pointbox}

\chapter{和・差・積および商の微分法}
\markboth{第6章}{和差積商の微分}
%==========================================================

\section{和と差}

$y=u+v$ なら、$\dd y=\dd u+\dd v$。それぞれの微分を足したり引いたりするだけです。

\section{積の微分}

$y=uv$ のとき、$u,v$ がそれぞれ $\dd u,\dd v$ だけ増えると、面積 $uv$ の長方形が、次のように大きくなります。

\begin{figure}[htbp]
\centering
\begin{tikzpicture}[scale=1.0]
 \draw[thick, fill=softblue] (0,0) rectangle (4,2.5);
 \draw[thick, fill=softamber] (4,0) rectangle (4.7,2.5);
 \draw[thick, fill=softamber] (0,2.5) rectangle (4,3.0);
 \draw[thick, fill=softrose] (4,2.5) rectangle (4.7,3.0);
 \node at (2,1.25) {$uv$};
 \node[font=\scriptsize] at (4.35,1.25) {$u\,\dd v$};
 \node[font=\scriptsize] at (2,2.75) {$v\,\dd u$};
 \draw[->] (5.5,2.75)--(4.75,2.75); \node[right, font=\tiny] at (5.5,2.75) {$\dd u\,\dd v$（無視）};
 \node[below, font=\small] at (2,0) {$u$}; \node[left, font=\small] at (0,1.25) {$v$};
\end{tikzpicture}
\caption{長方形の縦横が増えたときの面積の増え方。$uv\to uv+u\,\dd v+v\,\dd u+\dd u\,\dd v$。（見やすいよう増分を大きく描いている）}
\end{figure}

小さなものどうしのかけ算 $\dd u\,\dd v$ を捨てて、

\[
\dd(uv)=u\,\dd v+v\,\dd u.
\]

\begin{examplebox}{例}
$y=(x^2+1)(x^3-2)$。$u=x^2+1,\ v=x^3-2$ とおくと、$\dd u/\dd x=2x$、$\dd v/\dd x=3x^2$。\\
$\dfrac{\dd y}{\dd x}=(x^2+1)\cdot3x^2+(x^3-2)\cdot2x=3x^4+3x^2+2x^4-4x=5x^4+3x^2-4x$。
\end{examplebox}

\section{商の微分}

$y=u/v$ のとき、

\[
\frac{\dd y}{\dd x}=\frac{v\,\dfrac{\dd u}{\dd x}-u\,\dfrac{\dd v}{\dd x}}{v^2}.
\]

\begin{figure}[htbp]
\centering
\begin{tikzpicture}[node distance=7mm]
 \node[fbox, text width=2.2cm] (s) {和・差\\$\dd(u\pm v)$\\$=\dd u\pm\dd v$};
 \node[abox, text width=3.2cm, right=of s] (p) {積\\$\dd(uv)$\\$=u\,\dd v+v\,\dd u$};
 \node[gbox, text width=4.2cm, right=of p] (q) {商\\$\dd(u/v)$\\$=\dfrac{v\,\dd u-u\,\dd v}{v^2}$};
\end{tikzpicture}
\caption{3つの公式。積の公式が土台で、商は積の公式から導ける。}
\end{figure}

\begin{pointbox}{この章のポイント}
\begin{itemize}[leftmargin=1.5em]
\item 積の微分：（前を微分×後）＋（前×後を微分）。
\item 商の微分：（分子の微分×分母 $-$ 分子×分母の微分）$\div$ 分母の2乗。
\end{itemize}
\end{pointbox}

\section{微分をくり返す}

微分した結果をもう一度微分する。これを\textbf{逐次微分}（2階微分、3階微分…）といいます。$y=x^5$ なら、

\begin{center}

\toprule
$y$ & $=$ & $x^5$\\
$\dfrac{\dd y}{\dd x}$ & $=$ & $5x^4$\\
$\dfrac{\dd^2y}{\dd x^2}$ & $=$ & $20x^3$\\
$\dfrac{\dd^3y}{\dd x^3}$ & $=$ & $60x^2$\\
$\dfrac{\dd^4y}{\dd x^4}$ & $=$ & $120x$\\
$\dfrac{\dd^5y}{\dd x^5}$ & $=$ & $120$\\
$\dfrac{\dd^6y}{\dd x^6}$ & $=$ & $0$\\
\bottomrule
\end{tabular}
\end{center}

\section{2階微分は「傾きの変化」}

2階微分は、傾きがどう変わるかを表します。2階微分が正なら下に凸（谷のよう）、負なら上に凸（山のよう）。

\begin{itemize}[leftmargin=1.5em]

\chapter{時間が変化するとき（時間変化率）}

\section{速度と加速度}

位置 $s$ を時間 $t$ で微分すると速度、速度をもう一度微分すると加速度です。

\[
v=\frac{\dd s}{\dd t},\qquad a=\frac{\dd v}{\dd t}=\frac{\dd^2 s}{\dd t^2}.

物を真上に投げたとき（重力加速度を $g\approx9.8\ \mathrm{m/s^2}$）、初速 $v_0$ なら $s=v_0t-\frac12gt^2$。微分すると $v=v_0-gt$、さらに微分して $a=-g$（一定）。

\chapter{便利な巧妙法（合成関数の微分法・連鎖律）}
\markboth{第9章}{連鎖律}
%==========================================================

\section{つながった変化}

$y$ が $u$ の関数で、$u$ が $x$ の関数のとき、

\[
\frac{\dd y}{\dd x}=\frac{\dd y}{\dd u}\cdot\frac{\dd u}{\dd x}.
\]

これが連鎖律です。分数のように「$\dd u$ が約分される」と覚えると便利です（厳密な説明ではないが、計算の見通しはよい）。

\begin{figure}[htbp]
\centering
\begin{tikzpicture}[node distance=14mm]
 \node[fbox] (x) {$x$};
 \node[abox, right=of x] (u) {$u$};
 \node[gbox, right=of u] (y) {$y$};
 \draw[ax] (x)--node[above, lbl]{$\dd u/\dd x$} (u);
 \draw[ax] (u)--node[above, lbl]{$\dd y/\dd u$} (y);
 \draw[ax, crimson, bend left=40] (x) to node[above, lbl, text=crimson]{$\dd y/\dd x=\frac{\dd y}{\dd u}\cdot\frac{\dd u}{\dd x}$} (y);
\end{tikzpicture}
\caption{連鎖律。変化の「倍率」を、段ごとにかけていく。}
\end{figure}

\begin{examplebox}{例}
$y=(x^2+1)^3$。$u=x^2+1$ とおくと $y=u^3$。$\dd y/\dd u=3u^2$、$\dd u/\dd x=2x$。\\
$\dfrac{\dd y}{\dd x}=3(x^2+1)^2\cdot2x=6x(x^2+1)^2$。
\end{examplebox}

数値で確認：$x=1$ で、$y=8$。$x=1.01$ で $y=(1.0201+1)^3=2.0201^3\approx8.2436$。増加 $0.2436$、$\dd x=0.01$ なので比は約 $24.4$。公式の値は $6\cdot1\cdot2^2=24$。ほぼ一致します。

\begin{figure}[htbp]
\centering
\begin{tikzpicture}
\begin{axis}[width=9cm, height=5.5cm, xlabel={$x$}, xmin=0, xmax=2, ymin=0, ymax=30, grid=major, legend pos=north west, legend style={font=\footnotesize}]
 \addplot[thick, primaryblue, domain=0:2, samples=40] {(x^2+1)^3};
 \addplot[thick, crimson, dashed, domain=0:2, samples=40] {6*x*(x^2+1)^2};
 \legend{$y=(x^2+1)^3$,$\dd y/\dd x=6x(x^2+1)^2$}
\end{axis}
\end{tikzpicture}
\caption{関数とその微分。$x=1$ で微分の値は24。}
\end{figure}

\begin{pointbox}{この章のポイント}
\begin{itemize}[leftmargin=1.5em]
\item 連鎖律：$\dfrac{\dd y}{\dd x}=\dfrac{\dd y}{\dd u}\dfrac{\dd u}{\dd x}$。
\end{itemize}
\end{pointbox}

\chapter{極大値と極小値（最適化問題）}
\markboth{第11章}{極大・極小}
%==========================================================

\section{傾きが0になるところ}

山の頂上や谷底では、接線が水平、つまり $\dd y/\dd x=0$ です。このことを使って、「いちばん大きい」「いちばん小さい」を見つけます。

\begin{examplebox}{例：長さ20mの柵で、長方形の囲いを作る}
縦を $x$ とすると、横は $10-x$。面積 $A=x(10-x)=10x-x^2$。\\
$\dd A/\dd x=10-2x=0$ より $x=5$。面積は $5\times5=25\ \mathrm{m^2}$（正方形）。\\
2階微分は $-2<0$ なので上に凸、つまり極大。
\end{examplebox}

\centering

\end{axis}

\end{figure}

\section{もう1つの例：箱の体積}

縦横20cmの正方形の板の四隅から1辺 $x$ の正方形を切り取り、折り曲げてふたのない箱を作ります。体積は $V=x(20-2x)^2$。

\[
\frac{\dd V}{\dd x}=(20-2x)^2+x\cdot2(20-2x)(-2)=(20-2x)(20-6x).
\]

$\dd V/\dd x=0$ から $x=10$（箱にならない）または $x=\frac{10}{3}\approx3.33$。このとき $V=\frac{10}{3}\times\left(\frac{40}{3}\right)^2=\frac{16000}{27}\approx592.6\ \mathrm{cm^3}$。

\begin{itemize}[leftmargin=1.5em]
\item 最大・最小は $\dd y/\dd x=0$ を解く。
\item 2階微分の符号で、極大か極小かを判別。
\end{itemize}
\end{pointbox}

\chapter{曲線の曲率}
\markboth{第12章}{曲線の曲率}

\section{どれくらい曲がっているか}

曲線の曲がり具合は、その点にぴったり寄り添う円（曲率円）の半径で表せます。半径が小さいほど、きつく曲がっています。曲線 $y=f(x)$ の曲率半径は

\[
R=\frac{\bigl(1+(y')^2\bigr)^{3/2}}{|y''|}.
\]

\begin{examplebox}{例：放物線の原点での曲率円}
$y'=2x$、$y''=2$。原点では $y'=0$ なので、$R=\dfrac{1}{2}$。曲率円は、中心 $(0,\frac12)$、半径 $\frac12$ の円。
\end{examplebox}

\chapter{その他の便利な巧妙法（部分分数展開・逆関数など）}

\section{逆関数の微分}

$y=f(x)$ の逆関数 $x=g(y)$ について、傾きは逆数の関係です。

\[
\frac{\dd x}{\dd y}=\frac{1}{\;\dd y/\dd x\;}.
\]

\begin{examplebox}{例：2乗の関数と平方根}
$\dd y/\dd x=2x$ なので、$\dd x/\dd y=\dfrac1{2x}=\dfrac1{2\sqrt y}$。これは、$\sqrt y$ の微分の結果と一致します。
\end{examplebox}

\section{部分分数に分ける}

分数の式を、簡単な分数の和に分けると、微分や積分がやさしくなります。たとえば

\[
\frac{1}{x(x+1)}=\frac1x-\frac1{x+1}.

右辺を通分して確かめましょう。$\dfrac{(x+1)-x}{x(x+1)}=\dfrac1{x(x+1)}$。

\section{変数が2つ以上あるとき}

円柱の体積 $V=\pi r^2h$ は、半径 $r$ と高さ $h$ の2つの量で決まります。片方を固定して、もう片方だけ変えたときの変化の比が\textbf{偏微分}です。

\[
\frac{\partial V}{\partial r}=2\pi rh,\qquad\frac{\partial V}{\partial h}=\pi r^2.
\]

\centering

  \draw[<->, crimson] (1.4,0)--(1.4,3) node[midway, right, font=\small] {$h$};
  \draw[<->, accentgreen] (0,3)--(1,3) node[midway, above, font=\small] {$r$};
 \end{scope}
 \node[font=\small, text width=8cm, align=left] at (7.3,1.5) {$r$ を変える（$h$ は固定）：\\ 底面が広がる。$\partial V/\partial r=2\pi rh$。\\[2mm]$h$ を変える（$r$ は固定）：\\ 同じ底面の円盤を積み増す。$\partial V/\partial h=\pi r^2$。};
\end{tikzpicture}
\caption{円柱の体積を、$r$ と $h$ のどちらか一方だけ動かして調べる。}
\end{figure}

\section{全微分の見積もり}

$r=3,\ h=10$ のとき、$\dd r=0.1,\ \dd h=0.2$ だけ増やすと、体積の増加は

\[
\dd V\approx2\pi\cdot3\cdot10\cdot0.1+\pi\cdot3^2\cdot0.2=6\pi+1.8\pi=7.8\pi\approx24.5.
\]

実際の増加：$\pi(3.1^2\cdot10.2-3^2\cdot10)=\pi(98.022-90)=8.022\pi\approx25.2$。見積もりにかなり近い値です。

\item 偏微分：他の変数を固定して微分する。
\item 全微分：各変数の寄与を足して、全体の変化を見積もる。
\end{itemize}

\chapter{微分の逆演算としての積分計算}

\section{逆向きに戻す}

微分の公式を逆に読めば、積分の公式になります。

\[
\int x^n\,\dd x=\frac{x^{n+1}}{n+1}+C\qquad(n\ne-1).
\]

$C$ は積分定数です。微分すると消えてしまう定数があったからです。

\begin{center}
\begin{tabular}{ll}
\toprule
微分 & 積分（逆）\\
\midrule
$x^3\ \to\ 3x^2$ & $\displaystyle\int3x^2\dd x=x^3+C$\\
$x^2\ \to\ 2x$ & $\displaystyle\int x\,\dd x=\dfrac{x^2}{2}+C$\\
$e^x\ \to\ e^x$ & $\displaystyle\int e^x\dd x=e^x+C$\\
$\sin x\ \to\ \cos x$ & $\displaystyle\int\cos x\,\dd x=\sin x+C$\\
\bottomrule
\end{tabular}
\end{center}

 }
\end{axis}

\chapter{積分による面積の求値法（定積分）}
\markboth{第19章}{定積分と面積}

\section{$a$ から $b$ までの面積}

$a$ から $b$ までの面積は、原始関数 $F$（$F'=y$）を使って、

\[
\int_a^b y\,\dd x=F(b)-F(a).
\]

\begin{examplebox}{例1：2乗の曲線の下の面積}
$F(x)=\dfrac{x^3}{3}$。$F(3)-F(0)=9-0=9$。
\end{examplebox}

\begin{figure}[htbp]
\centering
\begin{tikzpicture}
\begin{axis}[width=9cm, height=5.5cm, xlabel={$x$}, ylabel={$y$}, xmin=0, xmax=3.5, ymin=0, ymax=10, grid=major]
 \addplot[thick, primaryblue, domain=0:3.5, samples=40] {x^2};
 \addplot[draw=none, fill=softblue, domain=0:3, samples=40] {x^2} \closedcycle;
 \node[font=\small] at (axis cs:2.1,3) {面積 $=9$};
\end{axis}
\end{tikzpicture}
\caption{$y=x^2$ の下の $0\le x\le3$ の面積は $9$。}
\end{figure}

\begin{examplebox}{例2：正弦曲線の下の面積}
$F(x)=-\cos x$。$F(\pi)-F(0)=(-\cos\pi)-(-\cos0)=1-(-1)=2$。
\end{examplebox}

\begin{figure}[htbp]
\centering
\begin{tikzpicture}
\begin{axis}[width=9.5cm, height=5.5cm, xlabel={$x$}, xmin=0, xmax=3.6, ymin=0, ymax=1.2, xtick={0,1.5708,3.1416}, xticklabels={$0$,$\frac\pi2$,$\pi$}, grid=major]

\end{axis}

\end{figure}

\begin{notebox}{注意}
面積が負になる場合があります。$y=\sin x$ を $\pi$ から $2\pi$ まで積分すると $-2$。グラフが $x$ 軸より下にあるときは、積分は「符号つきの面積」になります。
\end{notebox}

\begin{pointbox}{この章のポイント}
\begin{itemize}[leftmargin=1.5em]
\item 定積分 $=F(b)-F(a)$。
\item $x$ 軸より下の部分は、符号が負。
\end{itemize}
\end{pointbox}

\chapter{積分の巧妙法・落とし穴・そして勝利}
\markboth{第20章}{積分の巧妙法}
%==========================================================

\section{置換積分}

複雑な式は、新しい変数に置き換えると、やさしくなります。

\begin{examplebox}{例：$\displaystyle\int2x(x^2+1)^3\,\dd x$}
$u=x^2+1$ とおくと $\dd u=2x\,\dd x$。\quad $\displaystyle\int u^3\dd u=\frac{u^4}{4}+C=\frac{(x^2+1)^4}{4}+C$。
\end{examplebox}

検算：$\dfrac{\dd}{\dd x}\dfrac{(x^2+1)^4}{4}=(x^2+1)^3\cdot2x$ ✓。

\section{部分積分}

積の微分の公式 $\dd(uv)=u\,\dd v+v\,\dd u$ を逆に使うと、

\[
\int u\,\dd v=uv-\int v\,\dd u.
\]

\begin{figure}[htbp]
\centering
\begin{tikzpicture}[scale=1.0]
 \draw[->] (0,0)--(4.8,0) node[right] {$u$}; \draw[->] (0,0)--(0,3.6) node[above] {$v$};
 \draw[thick, primaryblue, domain=0.2:4.2, samples=40] plot (\x, {0.5+0.15*\x*\x});
 \fill[softblue, opacity=0.6] (0,0.5) -- plot[domain=0:3.5, samples=30] (\x, {0.5+0.15*\x*\x}) -- (3.5,0) -- (0,0) -- cycle;
 \draw[dashed] (3.5,0)--(3.5,{0.5+0.15*3.5*3.5}) -- (0,{0.5+0.15*3.5*3.5});
 \node[font=\footnotesize] at (1.6,0.6) {$\int v\,\dd u$};
 \node[font=\footnotesize] at (0.8,2.4) {$\int u\,\dd v$};
 \node[font=\footnotesize] at (3.9,-0.3) {$u$};
 \node[font=\footnotesize] at (-0.4,2.35) {$v$};
\end{tikzpicture}
\caption{部分積分の図。長方形 $uv$ を、曲線の下側（$\int v\,\dd u$）と左側（$\int u\,\dd v$）に分けて考える。合計は長方形全体の面積 $uv$。}
\end{figure}

\begin{examplebox}{例：$\displaystyle\int x e^x\dd x$}
$u=x,\ \dd v=e^x\dd x$ とすると $\dd u=\dd x,\ v=e^x$。\\
$\displaystyle\int xe^x\dd x=xe^x-\int e^x\dd x=xe^x-e^x+C=(x-1)e^x+C$。
\end{examplebox}

\section{落とし穴}

\begin{notebox}{よくある落とし穴}
\begin{itemize}[leftmargin=1.5em]
\item $\int x^n\dd x$ で $n=-1$ の場合は、$x^{n+1}/(n+1)$ が使えない（$\int\frac1x\dd x=\ln|x|+C$）。
\item 積分定数 $C$ を書き忘れる。
\item 定積分で、置換したら積分範囲も置き換える。
\end{itemize}
\end{notebox}

\begin{pointbox}{この章のポイント}
\begin{itemize}[leftmargin=1.5em]
\item 置換積分は連鎖律の逆。部分積分は積の微分の逆。
\end{itemize}
\end{pointbox}

\section{例：冷めるコーヒー}

ニュートンの冷却の法則：物体の温度と周囲の温度の差 $T$ は、その大きさに比例して減る。$\dd T/\dd t=-kT$。解は $T=T_0e^{-kt}$。

\begin{examplebox}{数値例}
最初の差が $60^\circ$C、$k=0.1$ 毎分とすると、10分後の差は $60e^{-1}\approx22.1^\circ$C。20分後は $60e^{-2}\approx8.1^\circ$C。
\end{examplebox}

\begin{itemize}[leftmargin=1.5em]

\item 現実の「比例して変わる」現象は、ほぼこの形。
\end{itemize}
\end{pointbox}
