\chapter{実変数}
\markboth{第1章}{実変数}
%==========================================================

数直線上の点は、小数で表せます。$\sqrt2=1.41421356\ldots$ は、有理数（分数）では表せない数です。ハーディは、実数を、有理数の「切り口」（デデキントの切断）として定義します。

\begin{figure}[htbp]
\centering
\begin{tikzpicture}
 \draw[ax] (-0.3,0)--(10.3,0);
 \foreach \xa/\lab in {0/1,10/2} {\draw (\xa,0.1)--(\xa,-0.1); \node[below] at (\xa,-0.1) {\lab};}
 \draw[thick, primaryblue, fill=softblue] (0,0.5) rectangle (4.14,0.9);
 \draw[thick, warmamber, fill=softamber] (4.14,0.5) rectangle (10,0.9);
 \draw[crimson, thick] (4.14,0.2)--(4.14,1.2);
 \node[above, font=\footnotesize] at (2.1,0.95) {$A$：$x^2<2$ の有理数};
 \node[above, font=\footnotesize] at (7.1,0.95) {$B$：$x^2>2$ の有理数};
 \node[below, font=\footnotesize, crimson] at (4.14,-0.3) {切り口 $=\sqrt2$};
\end{tikzpicture}
\caption{$\sqrt2$ を、有理数の2つの組 $A,B$ の切り口として定義する。切り口にあたる有理数はない。}
\end{figure}

\begin{figure}[htbp]
\centering
\begin{tikzpicture}[x=3.2cm]
 \foreach \l/\r/\yy in {1/2/0, 1.4/1.5/0.8, 1.41/1.42/1.6, 1.414/1.415/2.4} {

 }

\end{tikzpicture}
\caption{$\sqrt2$ を挟む区間を、どんどん狭めていく。（横軸は $x-1$。$1.41^2=1.9881<2<1.42^2=2.0164$ などで確認できる）}
\end{figure}

\begin{pointbox}{この章のポイント}
\begin{itemize}[leftmargin=1.5em]
\item 実数は、有理数の切り口（または区間の入れ子）として厳密に定義できる。
\end{itemize}
\end{pointbox}

\chapter{複素数}
\markboth{第3章}{複素数}
%==========================================================

複素数 $z=a+bi$ は、平面上の点 $(a,b)$ です。大きさ（絶対値）は $|z|=\sqrt{a^2+b^2}$。たとえば $z=3+4i$ なら $|z|=5$。

\begin{figure}[htbp]
\centering
\begin{tikzpicture}[scale=1.0]
 \draw[->] (-0.5,0)--(5.5,0) node[right] {実軸}; \draw[->] (0,-0.5)--(0,4.6) node[above] {虚軸};
 \draw[thick, primaryblue, ->] (0,0)--(1,2) node[above left, font=\small] {$1+2i$};
 \draw[thick, accentgreen, ->] (0,0)--(3,1) node[below right, font=\small] {$3+i$};
 \draw[thick, crimson, ->] (0,0)--(4,3) node[above right, font=\small] {$4+3i$};
 \draw[dashed] (1,2)--(4,3)--(3,1);
\end{tikzpicture}
\caption{複素数の足し算は、平行四辺形の対角線。$(1+2i)+(3+i)=4+3i$。}
\end{figure}

\begin{pointbox}{この章のポイント}
\begin{itemize}[leftmargin=1.5em]
\item 複素数は平面上の点。足し算は、ベクトルの和。
\end{itemize}
\end{pointbox}

\chapter{正の整数変数の関数の極限}
\markboth{第4章}{数列の極限}
%==========================================================

数列 $a_n=\dfrac1n$ は、$n$ が大きくなると $0$ に近づきます。「近づく」を厳密にいうと、「どんな小さな正の数 $\varepsilon$ に対しても、十分大きな $N$ があって、$n>N$ ならば $|a_n-0|<\varepsilon$」。

\begin{figure}[htbp]
\centering
\begin{tikzpicture}
\begin{axis}[width=10cm, height=5.5cm, xlabel={$n$}, ylabel={$a_n=1/n$}, xmin=0, xmax=30, ymin=-0.05, ymax=1.1, grid=major]
 \addplot[only marks, mark=*, mark size=1.5pt, color=primaryblue, samples at={1,...,30}] {1/x};
 \addplot[draw=none, fill=softrose, fill opacity=0.6] coordinates {(0,-0.1)(30,-0.1)(30,0.1)(0,0.1)} \closedcycle;
 \draw[dashed, crimson] (axis cs:10,-0.05)--(axis cs:10,1.1);
 \node[crimson, font=\footnotesize, anchor=south west] at (axis cs:10.4,0.4) {$N=10$};
\end{axis}
\end{tikzpicture}
\caption{$\varepsilon=0.1$ のとき、$n>10$ なら $1/n<0.1$。ピンクの帯の中に、$n>10$ のすべての点が入る。}
\end{figure}

もう1つの例は、$\left(1+\frac1n\right)^n$。$n=1$ で $2$、$n=10$ で $2.594$、$n=100$ で $2.705$、$n=1000$ で $2.717$。極限は $e=2.71828\ldots$ です。

\begin{pointbox}{この章のポイント}
\begin{itemize}[leftmargin=1.5em]
\item 極限：どんな $\varepsilon$ の帯にも、$n$ が十分大きければ入る。
\end{itemize}
\end{pointbox}

\chapter{連続変数の関数の極限・連続性}
\markboth{第5章}{関数の極限と連続}
%==========================================================

関数 $f(x)=2x+1$ が $x=1$ で連続であるとは、$f(1)=3$ のまわりの、どんな小さな範囲（$\varepsilon$）に対しても、$x=1$ のまわりに、十分小さな範囲（$\delta$）があって、その範囲の $x$ なら $f(x)$ が $3\pm\varepsilon$ に収まる、ということです。

\begin{figure}[htbp]
\centering
\begin{tikzpicture}
\begin{axis}[width=9cm, height=6cm, xlabel={$x$}, ylabel={$y$}, xmin=0.6, xmax=1.4, ymin=2.5, ymax=3.5, grid=major]
 \addplot[thick, primaryblue, domain=0.6:1.4] {2*x+1};
 \addplot[draw=none, fill=softamber, fill opacity=0.5] coordinates {(0.6,2.8)(1.4,2.8)(1.4,3.2)(0.6,3.2)} \closedcycle;
 \addplot[draw=none, fill=softgreen, fill opacity=0.6] coordinates {(0.9,2.5)(1.1,2.5)(1.1,3.5)(0.9,3.5)} \closedcycle;
 \addplot[only marks, mark=*, color=crimson] coordinates {(1,3)};
\end{axis}
\end{tikzpicture}
\caption{$\varepsilon=0.2$（黄の横帯 $2.8<y<3.2$）に対して、$\delta=0.1$（緑の縦帯 $0.9<x<1.1$）をとれば、その範囲の曲線は黄の帯に収まる。}
\end{figure}

\begin{pointbox}{この章のポイント}
\begin{itemize}[leftmargin=1.5em]
\item 連続性：$\varepsilon$ に対して、必ず $\delta$ が見つかる。
\end{itemize}
\end{pointbox}

\chapter{導関数と積分}
\markboth{第6章}{導関数と積分}
%==========================================================

導関数は、$\dfrac{f(x+h)-f(x)}{h}$ の $h\to0$ の極限、積分は、面積を細かい長方形で近似した極限です。$\int_0^1x^2\,\dd x=\dfrac13$ を、長方形の和で確かめましょう。

\begin{center}
\begin{tabular}{rrr}
\toprule
長方形の本数 $n$ & 左端の和（下から） & 右端の和（上から）\\
\midrule
10 & 0.285 & 0.385\\
100 & 0.32835 & 0.33835\\
真の値 & \multicolumn{2}{c}{$0.3333\ldots$}\\
\bottomrule
\end{tabular}
\end{center}

$n=10$ の左端の和は $\frac{1}{1000}(0^2+1^2+\cdots+9^2)=\frac{285}{1000}$、右端の和は $\frac{1}{1000}(1^2+\cdots+10^2)=\frac{385}{1000}$。

\begin{figure}[htbp]
\centering
\begin{tikzpicture}
\begin{axis}[width=9.5cm, height=6cm, xlabel={$x$}, ylabel={$y=x^2$}, xmin=0, xmax=1.1, ymin=0, ymax=1.1, grid=major]
 \addplot[thick, primaryblue, domain=0:1, samples=40] {x^2};
 \foreach \k in {0,...,9} {
   \addplot[draw=crimson, fill=softrose, fill opacity=0.5] coordinates {(\k/10,0)(\k/10+0.1,0)(\k/10+0.1,{(\k/10+0.1)^2})(\k/10,{(\k/10+0.1)^2})} \closedcycle;
 }
\end{axis}
\end{tikzpicture}
\caption{$y=x^2$ の下の面積を、右端の高さの10本の長方形で上から近似（合計 $0.385$）。本数を増やすと $1/3$ に近づく。}
\end{figure}

\begin{pointbox}{この章のポイント}
\begin{itemize}[leftmargin=1.5em]
\item 積分は、上からの和と下からの和で挟んで定義する。
\end{itemize}
\end{pointbox}

\chapter{微分積分学の補足的な定理}
\markboth{第7章}{補足的な定理}
%==========================================================

\textbf{平均値の定理}：なめらかな曲線の2点を結ぶ弦と平行な接線が、2点のあいだに必ず1本ある。$f(x)=x^2$ を $[0,2]$ で考えると、弦の傾きは $\frac{4-0}{2-0}=2$。接線の傾き $2x=2$ から $c=1$。

\begin{figure}[htbp]
\centering
\begin{tikzpicture}
\begin{axis}[width=8.5cm, height=6cm, xlabel={$x$}, ylabel={$y$}, xmin=-0.3, xmax=2.5, ymin=-0.5, ymax=4.5, grid=major]
 \addplot[thick, primaryblue, domain=0:2.3, samples=40] {x^2};
 \addplot[dashed, crimson, domain=0:2] {2*x};
 \addplot[thick, accentgreen, domain=0.2:2.2] {1+2*(x-1)};
 \addplot[only marks, mark=*, color=black] coordinates {(0,0)(2,4)(1,1)};
 \node[font=\footnotesize, anchor=south east] at (axis cs:1,1) {$c=1$};
\end{axis}
\end{tikzpicture}
\caption{平均値の定理。弦（赤）と平行な接線（緑）が $x=1$ にある。}
\end{figure}

\begin{pointbox}{この章のポイント}
\begin{itemize}[leftmargin=1.5em]
\item 平均値の定理：弦と平行な接線が、必ず区間の中にある。
\end{itemize}
\end{pointbox}

\chapter{無限級数と無限積分の収束}
\markboth{第8章}{収束}
%==========================================================

無限に足しても有限の値になる級数と、ならない級数があります。$\displaystyle\sum_{n=1}^\infty\frac1{n^2}=\frac{\pi^2}{6}\approx1.6449$ は収束し、$\displaystyle\sum\frac1n$ は発散します。

\begin{figure}[htbp]
\centering
\begin{tikzpicture}
\begin{axis}[width=10cm, height=5.5cm, xlabel={足した項の数 $n$}, ylabel={部分和}, xmode=log, xmin=1, xmax=1000, ymin=0, ymax=8, grid=major, legend pos=north west, legend style={font=\footnotesize}]
 \addplot[thick, primaryblue, mark=*] coordinates {(1,1)(2,1.25)(5,1.4636)(10,1.5498)(100,1.6350)(1000,1.6439)};
 \addplot[thick, crimson, mark=square*] coordinates {(1,1)(10,2.929)(100,5.187)(1000,7.485)};
 \addplot[dashed, gray, domain=1:1000] {1.6449};
 \legend{$\sum1/n^2$（収束）,$\sum1/n$（発散）}
\end{axis}
\end{tikzpicture}
\caption{2つの級数の部分和。$\sum1/n^2$ は $1.6449$ に頭打ち、$\sum1/n$ は対数的に増え続ける。}
\end{figure}

\begin{pointbox}{この章のポイント}
\begin{itemize}[leftmargin=1.5em]
\item 収束・発散は、部分和が一定の値に近づくかどうかで決まる。
\end{itemize}
\end{pointbox}

\end{itemize}
\end{pointbox}

\chapter{対数関数・指数関数・円関数の一般理論}
\markboth{第10章}{一般理論}
%==========================================================

複素数に指数関数を拡張すると、オイラーの公式 $e^{i\theta}=\cos\theta+i\sin\theta$ が現れます。$\theta=\pi$ とすると、$e^{i\pi}=-1$。指数関数と三角関数が、1つの式で結ばれます。

\begin{figure}[htbp]
